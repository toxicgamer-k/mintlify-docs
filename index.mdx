export const RotatingWord = ({ words }) => {
  const [index, setIndex] = useState(0);
  const [visible, setVisible] = useState(true);
  useEffect(() => {
    let swap;
    const timer = setInterval(() => {
      setVisible(false);
      swap = setTimeout(() => {
        setIndex((current) => (current + 1) % words.length);
        setVisible(true);
      }, 300);
    }, 2400);
    return () => {
      clearInterval(timer);
      clearTimeout(swap);
    };
  }, [words.length]);
  return (
    <span
      className={`inline-block transition-all duration-300 motion-reduce:transition-none ${visible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-2"}`}
    >
      {words[index]}
    </span>
  );
};
