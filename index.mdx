export const TrustedBy = () => {
  const [stats, setStats] = useState(null);

  useEffect(() => {
    let active = true;
    fetch("https://chirper.app/api/stats")
      .then((res) => (res.ok ? res.json() : null))
      .then((data) => {
        if (active && data && data.servers > 0 && data.members > 0) setStats(data);
      })
      .catch(() => {});
    return () => {
      active = false;
    };
  }, []);

  if (!stats) return <>Trusted by over 600,000 users.</>;

  const format = (n) => n.toLocaleString("en-US");
  const members = Math.floor(stats.members / 1000) * 1000;
  return (
    <>
      Trusted by <strong className="font-semibold">{format(stats.servers)}</strong> servers and{" "}
      <strong className="font-semibold">{format(members)}+</strong> members.
    </>
  );
};
